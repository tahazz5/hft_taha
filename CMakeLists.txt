cmake_minimum_required(VERSION 3.16)
project(orderbook LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(ORDERBOOK_SANITIZERS "Enable address and undefined behavior sanitizers" OFF)
add_library(orderbook INTERFACE)
target_include_directories(orderbook INTERFACE ${CMAKE_CURRENT_SOURCE_DIR}/include)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(orderbook INTERFACE -Wall -Wextra -Wpedantic)
  if(ORDERBOOK_SANITIZERS)
    target_compile_options(orderbook INTERFACE -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(orderbook INTERFACE -fsanitize=address,undefined)
  endif()
endif()
enable_testing()
add_executable(orderbook_tests tests/orderbook_tests.cpp)
add_executable(orderbook_benchmark benchmarks/orderbook_benchmark.cpp)
add_executable(orderbook_replay src/replay.cpp)
foreach(target orderbook_tests orderbook_benchmark orderbook_replay)
  target_link_libraries(${target} PRIVATE orderbook)
endforeach()
add_test(NAME orderbook_tests COMMAND orderbook_tests)
add_test(NAME replay_smoke COMMAND orderbook_replay ${CMAKE_CURRENT_SOURCE_DIR}/examples/orders.txt)
set_tests_properties(replay_smoke PROPERTIES PASS_REGULAR_EXPRESSION "TRADE 1 2 100 4")

# Binance connectivity is optional: the existing C++ engine has no new dependencies.
if(UNIX)
  add_executable(binance_marketdata src/binance_marketdata.cpp)
  target_link_libraries(binance_marketdata PRIVATE orderbook)
endif()
add_executable(market_depth_tests tests/market_depth_tests.cpp)
target_link_libraries(market_depth_tests PRIVATE orderbook)
add_test(NAME market_depth_tests COMMAND market_depth_tests)
find_package(Python3 3.11 COMPONENTS Interpreter QUIET)
if(Python3_Interpreter_FOUND)
  add_test(NAME binance_connector_tests COMMAND ${Python3_EXECUTABLE} -m unittest discover -s tests -p test_binance.py)
  set_tests_properties(binance_connector_tests PROPERTIES WORKING_DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR})
  if(UNIX)
    add_test(NAME binance_bridge_tests COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_binance_bridge.py $<TARGET_FILE:binance_marketdata>)
    set_tests_properties(binance_bridge_tests PROPERTIES WORKING_DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR})
  endif()
endif()
